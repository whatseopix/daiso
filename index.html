function doGet(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. 참가자 시트 확인 const pSheet = ss.getSheetByName("참가자"); let participants = []; if (!pSheet) { throw new Error("['참가자'] 시트를 찾을 수 없습니다. 시트 이름을 확인해주세요."); } const pValues = pSheet.getDataRange().getValues(); for (let i = 1; i < pValues.length; i++) { if (pValues[i][0] !== "" && pValues[i][0] !== null) { participants.push(String(pValues[i][0]).trim()); } } // 2. 상품 시트 확인 const gSheet = ss.getSheetByName("상품"); let gifts = []; if (!gSheet) { throw new Error("['상품'] 시트를 찾을 수 없습니다. 시트 이름을 확인해주세요."); } const gValues = gSheet.getDataRange().getValues(); for (let i = 1; i < gValues.length; i++) { const num = gValues[i][0]; const name = gValues[i][1]; const selectedVal = gValues[i][2]; const empName = gValues[i][3]; if (num !== "" && num !== null) { const isSelected = (selectedVal === true || selectedVal === 'O' || selectedVal === 'o' || (selectedVal !== "" && selectedVal !== null && selectedVal !== false)); gifts.push({ number: num, name: name || "상품", selected: isSelected, empName: empName || "" }); } } const result = { participants: participants, gifts: gifts }; return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ error: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { const lock = LockService.getScriptLock(); try { // 동시성 제어: 동시에 요청이 들어와도 순차적으로 처리하도록 락(Lock) 획득 (최대 10초 대기) if (!lock.tryLock(10000)) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: '요청이 너무 많습니다. 잠시 후 다시 시도해주세요.' })) .setMimeType(ContentService.MimeType.JSON); } const contents = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); if (contents.action === 'reset') { if (contents.password === '1234') { const gSheet = ss.getSheetByName("상품"); const lastRow = gSheet.getLastRow(); if (lastRow > 1) { gSheet.getRange(2, 3, lastRow - 1, 3).clearContent(); } lock.releaseLock(); return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON); } else { lock.releaseLock(); return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: '비밀번호가 틀렸습니다.' })) .setMimeType(ContentService.MimeType.JSON); } } const giftNumber = contents.giftNumber; const name = contents.name; const gSheet = ss.getSheetByName("상품"); const gValues = gSheet.getDataRange().getValues(); let targetRow = -1; let isAlreadySelected = false; for (let i = 1; i < gValues.length; i++) { if (String(gValues[i][0]) === String(giftNumber)) { targetRow = i + 1; const selectedVal = gValues[i][2]; isAlreadySelected = (selectedVal === true || selectedVal === 'O' || selectedVal === 'o' || (selectedVal !== "" && selectedVal !== null && selectedVal !== false)); break; } } if (targetRow === -1) { lock.releaseLock(); return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: '존재하지 않는 상품 번호입니다.' })) .setMimeType(ContentService.MimeType.JSON); } // 서버 단에서 이미 선택되었는지 철저하게 재검증 if (isAlreadySelected) { lock.releaseLock(); return ContentService.createTextOutput(JSON.stringify({ status: 'already_selected' })) .setMimeType(ContentService.MimeType.JSON); } // 정상 선택 기록 const today = Utilities.formatDate(new Date(), "Asia/Seoul", "yyyy. M. d"); gSheet.getRange(targetRow, 3).setValue("O"); gSheet.getRange(targetRow, 4).setValue(name); gSheet.getRange(targetRow, 5).setValue(today); lock.releaseLock(); return ContentService.createTextOutput(JSON.stringify({ status: 'success' })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { if (lock.hasLock()) { lock.releaseLock(); } return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
